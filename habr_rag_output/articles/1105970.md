Код от ИИ

```
#include <cairo.h>
int main() {
    // 1. Создаем поверхность для рисования (изображение 400x300) и контекст cairo
    cairo_surface_t *surface = cairo_image_surface_create(CAIRO_FORMAT_ARGB32, 400, 300);
    cairo_t *cr = cairo_create(surface);
    // Заливаем фон белым цветом
    cairo_set_source_rgb(cr, 1.0, 1.0, 1.0);
    cairo_paint(cr);
    // Параметры нашего прямоугольника
    double x = 50.0, y = 50.0;
    double width = 300.0, height = 200.0;
    // --- ЭТАП 1: ЗАЛИВКА ПРЯМОУГОЛЬНИКА ---
    // Создаем горизонтальный линейный градиент для заливки (слева направо)
    cairo_pattern_t *fill_pattern = cairo_pattern_create_linear(x, y, x + width, y);
    cairo_pattern_add_color_stop_rgb(fill_pattern, 0.0, 0.1, 0.4, 0.8); // Синий
    cairo_pattern_add_color_stop_rgb(fill_pattern, 1.0, 0.8, 0.1, 0.1); // Красный
    // Добавляем прямоугольник в текущий векторный контур
    cairo_rectangle(cr, x, y, width, height);
    // Устанавливаем паттерн заливки как источник и заполняем контур
    // (cairo_fill сохраняет контур пустым после выполнения, поэтому для обводки зададим его заново)
    cairo_set_source(cr, fill_pattern);
    cairo_fill(cr);
    // Освобождаем память паттерна заливки
    cairo_pattern_destroy(fill_pattern);
    // --- ЭТАП 2: ГРАДИЕНТНАЯ ОБВОДКА ---
    // Создаем вертикальный линейный градиент для обводки (сверху вниз)
    cairo_pattern_t *stroke_pattern = cairo_pattern_create_linear(x, y, x, y + height);
    cairo_pattern_add_color_stop_rgb(stroke_pattern, 0.0, 1.0, 0.8, 0.0); // Желтый
    cairo_pattern_add_color_stop_rgb(stroke_pattern, 0.5, 0.0, 0.8, 0.2); // Зеленый
    cairo_pattern_add_color_stop_rgb(stroke_pattern, 1.0, 0.5, 0.0, 0.8); // Фиолетовый
    // Снова добавляем прямоугольник в контур для выполнения обводки
    cairo_rectangle(cr, x, y, width, height);
    // Настраиваем толщину линии обводки
    cairo_set_line_width(cr, 12.0);
    // Устанавливаем паттерн обводки как источник и рисуем границы
    cairo_set_source(cr, stroke_pattern);
    cairo_stroke(cr);
    // Освобождаем память паттерна обводки
    cairo_pattern_destroy(stroke_pattern);
    // 2. Сохраняем полученный результат в PNG-файл
    cairo_surface_write_to_png(surface, "gradient_rect.png");
    // 3. Очищаем основные ресурсы cairo
    cairo_destroy(cr);
    cairo_surface_destroy(surface);
    return 0;
}
```

Теперь каждый желающий может протестировать результат, запустив [на облачной демонстрации](https://cloudtk-app.tcl-lang.org/cloudtk/VNC?session=new&Tk=SVG-widgets-Part-I) следующий пример:

```
package require tkpath
set t ".c_over"
toplevel $t
wm geometry $t 900x550
tkp::canvas $t.c -bg yellow
pack $t.c -fill both -expand 1 -padx 1m -pady 1m
set unit "userspace"
$t.c gradient create linear -units $unit -lineartransition [list 900 0 0 90] -stops {{0 red 1} {1 gray80 1}}
$t.c gradient create linear -units $unit -lineartransition [list 90 0 0 90] -stops {{0 cyan 1} {1 chocolate 1}}

$t.c gradient create linear -units $unit -lineartransition [list 40 40 300 300] -stops {{0 red 1} {0.5 gray80 1} {1 cyan 1}}
set myGradient [$t.c gradient create linear -units $unit -stops {{0 "gray80"} {1 "blue"}}]
set rgr [$t.c gradient create radial -units bbox -stops {{0 cyan 1} {0.5 gray80 1} {0.75 red 1} {1 snow 1}}]
$t.c create prect 80 240 280 340 -fill $rgr -stroke gradient1 -strokewidth 10 -rx 6

$t.c create pline 20 40 280 300 -strokewidth 20 -stroke gradient2
$t.c create ptext 100 80 -text "Stroke and Fill gradient" -fontsize 70 -fontweight bold -stroke gradient0 -strokewidth 15 -fill gradient1 -filloverstroke 1
$t.c create ptext 160 140 -text "filloverstroke=0" -fontsize 70 -fontweight bold -stroke red -strokewidth 8 -fill black -filloverstroke 0
$t.c create ptext 220 200 -text "filloverstroke=1" -fontsize 70 -fontweight bold -stroke red -strokewidth 8 -fill black -filloverstroke 1

lassign "40 200 40 200" x0 x1 xs xe
for {set i 1} {$i < 5} {incr i}  {
    set obj$i [$t.c create prect $xs 400 $xe 520 -stroke {} -strokewidth 20 -fill cyan -stroke red]
    $t.c create ptext $xs 400 -textanchor nw -text "-fill over stroke" -fontsize 20 -stroke black
    incr xs $x1
    incr xe $x1
}
$t.c itemconfigure $obj1 -stroke {}
$t.c itemconfigure [expr {$obj1 + 1}] -text "-stroke {}"
$t.c itemconfigure $obj2 -filloverstroke 0
$t.c itemconfigure [expr {$obj2 + 1}] -text "-filloverstroke 0\n-strokewidth 20"
$t.c itemconfigure $obj3 -filloverstroke 1
$t.c itemconfigure [expr {$obj3 + 1}] -text "-filloverstroke 1\n-strokewidth 20"
$t.c itemconfigure $obj4 -fillopacity 0.8 -strokeopacity 0.5
$t.c itemconfigure [expr {$obj4 + 1}] -text "-filloverstroke 0\n-strokewidth 20\n-fillopacity 0.8\n-strokeopacity 0.5"
```
